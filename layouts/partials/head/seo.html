{{- /* head/seo.html — title, meta description, canonical, robots, Open Graph, Twitter card, and RSS alternate. Self-contained; no theme dependency. */ -}} {{- $siteTitle := site.Title -}} {{- with site.Params.sitename }}{{ $siteTitle = . }}{{ end -}} {{- /* Page title: home uses the site name; inner pages prefix the page title. */ -}} {{- $title := "" -}} {{- if .IsHome -}} {{- $title = $siteTitle -}} {{- else -}} {{- $title = printf "%s · %s" .Title $siteTitle -}} {{- end -}} {{- /* Description: explicit .Description, else summary, else site description. The home page prefers the curated site description over an auto-summary (its rendered content makes a poor meta description). */ -}} {{- $desc := "" -}} {{- with .Description -}} {{- $desc = . -}} {{- else -}} {{- if .IsHome -}} {{- $desc = site.Params.description -}} {{- else -}} {{- with .Summary -}} {{- $desc = . -}} {{- else -}} {{- $desc = site.Params.description -}} {{- end -}} {{- end -}} {{- end -}} {{- $desc = $desc | plainify | htmlUnescape | strings.TrimSpace -}} {{- $desc = $desc | truncate 200 -}} {{- /* OG image: post cover image if present, else the hero image. Absolute URL. */ -}} {{- $ogImage := partial "og-default.html" . -}} {{- with .Params.cover -}}{{ $ogImage = . | absURL -}}{{ end -}} {{- with .Params.image -}}{{ $ogImage = . | absURL -}}{{ end -}} {{- $ogType := "website" -}} {{- if and .IsPage (eq .Section "posts") -}}{{ $ogType = "article" -}}{{ end -}} {{ $title }} {{- /* Open Graph */ -}} {{- if eq $ogType "article" }} {{- end }} {{- /* Twitter */ -}} {{- /* RSS alternate */ -}} {{- with .OutputFormats.Get "rss" }} {{- end }}
